cmake_minimum_required(VERSION 3.8)
project(transport_orthogonal_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_util
  pluginlib geometry_msgs nav_msgs tf2_ros)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/planner.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${dependencies})
pluginlib_export_plugin_description_file(nav2_core planner_plugin.xml)
install(TARGETS ${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES planner_plugin.xml DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_search test/test_search.cpp)
  target_include_directories(test_search PRIVATE include)
  add_test(NAME orthogonal_search COMMAND test_search)
endif()
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_export_dependencies(${dependencies})
ament_package()
